procedimento obstaculos_init(canvas_simulado : inteiro)
var
    novo_obs : TObstaculo
    i : inteiro
    slot_livre : inteiro
inicio
    // Encontra um slot livre no vetor
    slot_livre <- 0
    para i de 1 ate 100 faca
        se (obstaculos[i].ativo = FALSO) entao
            slot_livre <- i
            i <- 101  // Sai do loop
        fimse
    fimpara
    
    se (slot_livre > 0) entao
        // self.position = random.randint(0,750)
        novo_obs.posicao <- random_int(0, 750)
        
        // self.tipe = "Car"
        novo_obs.tipo <- "Car"
        
        // self.ost = self.canvas.create_rectangle(self.position,50,self.position+50,0,fill="gray")
        novo_obs.ost <- slot_livre  // Simula um ID único
        novo_obs.ativo <- VERDADEIRO
        
        // Armazena o obstáculo
        obstaculos[slot_livre] <- novo_obs
        num_obs <- num_obs + 1
    fimse
fimprocedimento

// FUNÇÃO: verificar_colisao 

funcao verificar_colisao() : logico
var
    carro_esq : inteiro
    carro_dir : inteiro
    carro_cima : inteiro
    carro_baixo : inteiro
    i : inteiro
    obst_esq : inteiro
    obst_dir : inteiro
    obst_cima : inteiro
    obst_baixo : inteiro
inicio
    carro_esq <- carro_x - carro_tamanho
    carro_dir <- carro_x + carro_tamanho
    carro_cima <- carro_y - carro_tamanho
    carro_baixo <- carro_y + carro_tamanho
    
    para i de 1 ate 100 faca
        se (obstaculos[i].ativo = VERDADEIRO) entao
            // if item != Carro and item != cenario:
            // Como não temos os IDs reais, comparamos pelo tipo e posição
            se (i <> carro_id e i <> cenario_id) entao
                // obst_pos = canvas.coords(item)
                // Simula coordenadas do obstáculo (posicao, 50, posicao+50, 0)
                obst_esq <- obstaculos[i].posicao
                obst_dir <- obstaculos[i].posicao + 50
                obst_cima <- 50
                obst_baixo <- 0
                
                // if (carro_pos[0] < obst_pos[2] and 
                //     carro_pos[2] > obst_pos[0] and 
                //     carro_pos[1] < obst_pos[3] and 
                //     carro_pos[3] > obst_pos[1]):
                se (carro_esq < obst_dir e 
                    carro_dir > obst_esq e 
                    carro_cima < obst_baixo e 
                    carro_baixo > obst_cima) entao
                    
                    retorne VERDADEIRO
                fimse
            fimse
        fimse
    fimpara
    
    retorne FALSO
fimfuncao

procedimento loop_principal()
var
    obst_id : inteiro
    contador : inteiro
inicio
    contador <- 0
    
    enquanto (jogo_ativo = VERDADEIRO) faca
        obstaculos_init(1)  
        

        para obst_id de 1 ate 100 faca
            se (obstaculos[obst_id].ativo = VERDADEIRO) entao
                obstaculos_descer(obst_id, 1)  
            fimse
        fimpara
        
        contador <- contador + 1
        

        se (verificar_colisao() = VERDADEIRO) entao

            carro_id <- 0  
            

            escreval("Game Over!")
            

            jogo_ativo <- FALSO
        fimse


        se (contador >= 10) entao
            contador <- 0
        fimse
    fimenquanto
fimprocedimento
